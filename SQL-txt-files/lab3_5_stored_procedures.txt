--since both borrowing and returning a book only modify the tables, and dont return anything, that means we can make them as procedures, no need for a function

create procedure borrow_a_book (book_id_arg integer, borrower_id_arg integer, return_date_arg date)
    language plpgsql
    as $$
    begin
        if not exists (select book_id from books where book_id = book_id_arg) then
            raise exception 'Book does not exist.';
        end if;
        if not exists (select borrower_id from borrowers where borrower_id = borrower_id_arg) then
            raise exception 'Borrower does not exist.';
        end if;
        insert into loans (book_id, borrower_id, loan_date, return_date) values
            (book_id_arg, borrower_id_arg, current_date, return_date_arg);
        raise notice 'Successful loan.';
    end $$;


create procedure return_a_book(book_id_arg integer, borrower_id_arg integer)
    language plpgsql
    as $$
    begin
    if not exists (select book_id from books where book_id = book_id_arg) then
        raise exception 'Book does not exist.';
    end if;
    if not exists (select borrower_id from borrowers where borrower_id = borrower_id_arg) then
        raise exception 'Borrower does not exist.';
    end if;
    update loans
        set returned = true, return_date = current_date
        where book_id = book_id_arg and borrower_id = borrower_id_arg and returned = false;
    raise notice 'Successful return.';
    end $$;

create or replace procedure borrow_a_book (book_id_arg integer, borrower_id_arg integer, return_date_arg date)
    language plpgsql
as $$
begin
    if not exists (select book_id from books where book_id = book_id_arg) then
        raise exception 'Book does not exist.';
    end if;
    if not exists (select borrower_id from borrowers where borrower_id = borrower_id_arg) then
        raise exception 'Borrower does not exist.';
    end if;
    if exists (select loans.loan_id from loans where book_id = book_id_arg and borrower_id = borrower_id_arg and returned is false) then
        raise exception 'Loan already exists! What are you waiting for to return that book?';
    end if;
    insert into loans (book_id, borrower_id, loan_date, return_date) values
        (book_id_arg, borrower_id_arg, current_date, return_date_arg);
    raise notice 'Successful loan.';
end $$;


create or replace procedure return_a_book(loan_id_arg integer)
    language plpgsql
as $$
begin
    if not exists (select loan_id from loans where loan_id = loan_id_arg) then
        raise exception 'Loan does not exist.';
    end if;
    if not exists (select loans.loan_id from loans where loan_id = loan_id_arg and returned is false) then
        raise exception 'Book already returned!';
    end if;
    update loans
    set returned = true, return_date = current_date
    where loan_id = loan_id_arg and returned = false;
    raise notice 'Successful return.';
end $$;
